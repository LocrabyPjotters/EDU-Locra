Voor Ontwikkelaars

Eén regel code aanpassen. Klaar.

Gebruik je bestaande OpenAI of Claude library. Pas alleen de base_url aan naar onze Nederlandse gateway. 

from openai import OpenAI

# Wijs direct naar de soevereine Nederlandse gateway van Locra PrivAI™
client = OpenAI(
    base_url="https://api.privai.locra.nl/v1",
    api_key="privai_live_8f3d91c7b5204ef8a961023d8c11e40a"
)

response = client.chat.completions.create(
    model="locra-privai-pro",
    messages=[
        {"role": "system", "content": "Je bent een didactische tutor."},
        {"role": "user", "content": "Leg kwantummechanica uit aan 5 VWO."}
    ],
    stream=True
)

for chunk in response:
    print(chunk.choices[0].delta.content or "", end="", flush=True)
import Anthropic from '@anthropic-ai/sdk';

// Drop-in compatibel met Claude berichten
const anthropic = new Anthropic({
  baseURL: 'https://api.privai.locra.nl',
  apiKey: 'privai_live_8f3d91c7b5204ef8a961023d8c11e40a'
});

const message = await anthropic.messages.create({
  model: 'locra-privai-pro',
  max_tokens: 1024,
  messages: [{ role: 'user', content: 'Beoordeel deze natuurkundige stelling.' }]
});

console.log(message.content[0].text);
